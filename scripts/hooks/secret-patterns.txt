# Credential patterns for .memory/ (POSIX extended regex, matched case-insensitively).
# One pattern per line; lines starting with # are ignored.
# Shared by scripts/hooks/pre-commit (commit-time) and scripts/hooks/memory-guard.sh (Claude Code PreToolUse).
(^|[^a-z0-9])sk-
# key=value / key: value forms, but not placeholders that point to an env var
# (e.g. spring.datasource.password=${DB_PASSWORD} is allowed).
(password|passwd|secret|token|api[_-]?key|apikey)[[:space:]]*[=:][[:space:]]*[^${[:space:]`]
# A word that contains key/token/secret/pass AND ends in 4+ digits looks like a value, not a variable name
# (e.g. TEST_API_KEY_DO_NOT_STORE_12345, FakeDbPass-Test-9876; but not ANTHROPIC_API_KEY or DB_PASSWORD).
[a-z0-9_-]*(key|token|secret|pass)[a-z0-9_-]*[0-9]{4,}
# Private key blocks and common provider prefixes
-----BEGIN [a-z ]*PRIVATE KEY-----
(^|[^a-z0-9])(ghp_|xox[bpas]-|AKIA)[a-z0-9]
